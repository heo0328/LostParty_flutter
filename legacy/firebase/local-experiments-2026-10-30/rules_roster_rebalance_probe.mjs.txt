function token(e) { return e.role == 'support' ? 'S' : 'D' + e.classNameSnapshot; }
function tokensAt(ids,es,i) { return i < ids.size() ? [es[ids[i]].role == 'support' ? 'S' : 'D' + es[ids[i]].classNameSnapshot] : []; }
function tokens(ids,es) { return tokensAt(ids,es,0).concat(tokensAt(ids,es,1)).concat(tokensAt(ids,es,2)).concat(tokensAt(ids,es,3)); }
function validParty(ts) { return ts.size() <= 4 && ts.toSet().size() == ts.size() && ts.removeAll(['S']).size() <= 3; }
function proofMember(ids,es,p,i,r,e) { return i >= ids.size() || (es[ids[i]].partyIndex == p
  && (ids[i] == r.commandPlayerId || es[ids[i]].characterId != e.characterId)); }
function proofParty(ids,es,p,r,e) { return ids is list && ids.size() <= 4
  && proofMember(ids,es,p,0,r,e) && proofMember(ids,es,p,1,r,e) && proofMember(ids,es,p,2,r,e) && proofMember(ids,es,p,3,r,e); }
function proofShape(w,es,r,e) { return w is map && w.keys().hasOnly(['p0','p1','swap'])
  && proofParty(w.p0,es,0,r,e) && proofParty(w.p1,es,1,r,e) && w.swap is list
  && w.p0.concat(w.p1).size() == r.memberPlayerIds.size()
  && w.p0.concat(w.p1).toSet() == r.memberPlayerIds.toSet(); }
function slotAt(ids,es,i) { return i < ids.size() ? [es[ids[i]].slotIndex] : []; }
function firstFree(ids,es) { let used = slotAt(ids,es,0).concat(slotAt(ids,es,1)).concat(slotAt(ids,es,2)).concat(slotAt(ids,es,3));
  return !(0 in used) ? 0 : (!(1 in used) ? 1 : (!(2 in used) ? 2 : (!(3 in used) ? 3 : -1))); }
// Remove exactly one occurrence, including a temporarily duplicated dealer class.
function removeOne(ts,t) { let rest = ts.removeAll([t]); let count = ts.size() - rest.size();
  return rest.concat(count == 1 ? [] : (count == 2 ? [t] : (count == 3 ? [t,t] : [t,t,t]))); }
function swapValid(w,es,r,e,a,b) {
  let left = w.swap[0]; let right = w.swap[1];
  let le = left == r.commandPlayerId ? e : es[left];
  let re = right == r.commandPlayerId ? e : es[right];
  let lt = token(le); let rt = token(re);
  return left != right && left != r.leaderPlayerId && right != r.leaderPlayerId
    && es[left].partyIndex == 0 && es[right].partyIndex == 1
    && validParty(removeOne(a,lt).concat([rt])) && validParty(removeOne(b,rt).concat([lt]));
}
function placementDecision(w,es,r,e) {
  let old = es[r.commandPlayerId]; let t = token(e);
  let a = tokens(w.p0.removeAll([r.commandPlayerId]),es);
  let b = tokens(w.p1.removeAll([r.commandPlayerId]),es);
  let same = e.partyIndex == old.partyIndex && e.slotIndex == old.slotIndex;
  let ipA = a.concat(old.partyIndex == 0 ? [t] : []);
  let ipB = b.concat(old.partyIndex == 1 ? [t] : []);
  let move = r.commandPlayerId != r.leaderPlayerId
    && validParty(a.concat(old.partyIndex == 1 ? [t] : []))
    && validParty(b.concat(old.partyIndex == 0 ? [t] : []));
  return validParty(ipA) && validParty(ipB) ? (same && w.swap.size() == 0)
    : (move ? (w.swap.size() == 0 && e.partyIndex == 1 - old.partyIndex
      && e.slotIndex == firstFree(old.partyIndex == 0 ? w.p1 : w.p0,es))
    : (same && w.swap.size() == 2 && swapValid(w,es,r,e,ipA,ipB)));
}
function positionFresh(id,r) { return r.updatedAt == request.time && r.revision == get(roomPath(id)).data.revision + 1; }
function positionEntry(before,after,p,s) { return after.diff(before).affectedKeys().hasOnly(['partyIndex','slotIndex','updatedAt'])
  && after.partyIndex == p && after.slotIndex == s && after.updatedAt == request.time; }
function positionSource(id,b,r) {

    && request.time >= b.cycleStartedAt && request.time < b.cycleEndsAt && positionFresh(id,r)
    && r.diff(b).affectedKeys().hasOnly(['revision','rosterRevision','commandId','commandType','commandPlayerId','updatedAt','positionTarget','placementProof'])
    && r.rosterRevision == b.rosterRevision + 1 && old.revision == b.rosterRevision && next.revision == r.rosterRevision
    && q.keys().hasOnly(['partyIndex','slotIndex','other']) && q.partyIndex is int && q.slotIndex is int
    && q.partyIndex >= 0 && q.partyIndex < b.partyCount && q.slotIndex >= 0 && q.slotIndex < 4
    && r.commandPlayerId != b.leaderPlayerId && (q.partyIndex != e.partyIndex || q.slotIndex != e.slotIndex)
    && next.entries.diff(old.entries).affectedKeys() == (q.other == null ? [r.commandPlayerId] : [r.commandPlayerId,q.other]).toSet()
    && positionEntry(e,next.entries[r.commandPlayerId],q.partyIndex,q.slotIndex)
    && (q.other == null ? true : (q.other != b.leaderPlayerId && q.other != r.commandPlayerId
      && old.entries[q.other].partyIndex == q.partyIndex && old.entries[q.other].slotIndex == q.slotIndex
      && positionEntry(old.entries[q.other],next.entries[q.other],e.partyIndex,e.slotIndex)))
    && eventRequired(r);
}
function uniqueSlots(ids,es) { let ss = slotAt(ids,es,0).concat(slotAt(ids,es,1)).concat(slotAt(ids,es,2)).concat(slotAt(ids,es,3));
  return ss.toSet().size() == ss.size(); }
function positionLayout(es,r) { let w = r.placementProof;
  return w.swap.size() == 0
    && validParty(tokens(w.p0,es)) && validParty(tokens(w.p1,es))
    && uniqueSlots(w.p0,es) && uniqueSlots(w.p1,es);
}
