  memberCount:1,dealerCount:1,supportCount:0,maxMemberCount:8,hasAvailableSlot:true,manualRecruitmentEnabled:true,partyCount:2,membersPerParty:4,

async function admin(work){await env.withSecurityRulesDisabled(async ctx=>{const db=ctx.firestore(),b=writeBatch(db);work(b,db);await b.commit();});}
async function seed({generation=1,state=null,storedClaim=null,assignment=false,character={},account={},binding='active'}={}){
  await env.clearFirestore();await admin((b,db)=>{
    b.set(doc(db,'authBindings/u0'),{status:binding,playerId:'p0'});b.set(doc(db,'authBindings/u1'),{status:'active',playerId:'p1'});
    b.set(doc(db,`characters/${cid}`),{status:'active',ownerPlayerId:'p0',ownershipRevision:generation,gameAccountId:'a0',characterName:'name',characterClassName:'class',...character});
    b.set(doc(db,'gameAccounts/a0'),{status:'active',ownerPlayerId:'p0',ownershipRevision:generation,...account});
    if(state)b.set(doc(db,statePath),state);if(storedClaim)b.set(doc(db,claimPath),storedClaim);
    if(assignment)b.set(doc(db,assignmentPath),{playerId:'p0',characterId:cid,raidId:'raid',ownershipRevision:1});
  });
}
async function snapshot(){let value;await env.withSecurityRulesDisabled(async ctx=>{
  value=await Promise.all([statePath,claimPath,assignmentPath,'recruitments/room','recruitments/room/roster/current','partyEvents/partyCreated_room_1']
    .map(async p=>[p,(await getDoc(doc(ctx.firestore(),p))).data()??null]));});return value;}
function batch(actor='u0'){const db=actor===null?env.unauthenticatedContext().firestore():env.authenticatedContext(actor).firestore();return {db,b:writeBatch(db)};}
function create(patch={},actor='u0',path=claimPath){const {db,b}=batch(actor);b.set(doc(db,path),{...claim(),...patch});return b.commit();}
function remove(actor='u0'){const {db,b}=batch(actor);b.delete(doc(db,claimPath));return b.commit();}
async function denied(work){const before=await snapshot();await assertRulesDenied(work());assert.deepEqual(await snapshot(),before);}
async function check(name,work){if(process.env.ROSTER_CLAIM_RACE && !name.startsWith('concurrent certification'))return;
  try{await work();results.push({name,passed:true});}catch(e){results.push({name,passed:false,error:e.message});throw e;}
  finally{fs.writeFileSync(`${directory}/${process.env.ROSTER_CLAIM_RACE?'race':baseline?'baseline':'claim'}-results.json`,JSON.stringify({sha256,results},null,2));}}
async function load(content){const response=await fetch('http://127.0.0.1:8080/emulator/v1/projects/demo-party-group-cutover:securityRules',{
  method:'PUT',headers:{'Content-Type':'application/json'},body:JSON.stringify({rules:{files:[{content}]}})});
  assert.equal(response.status,200);const body=await response.json();assert.deepEqual(body,{});}

function replacement(withClaim=false,retain={}){const {db,b}=batch(),t=serverTimestamp(),r={...room(t),currentCycleKey:'2026-10-08'};
  b.set(doc(db,'recruitments/room'),r);b.set(doc(db,'recruitments/room/roster/current'),{schemaVersion:1,recruitmentId:'room',revision:1,entries:{p0:entry(t)}});
  b.set(doc(db,statePath),{schemaVersion:1,playerId:'p0',characterId:cid,ownershipRevision:1,recruitmentId:'room',raidId:'raid',difficultyId:'hard',
    cycleKey:r.currentCycleKey,role:'dealer',status:'active',source:'party',revision:1,createdAt:t,updatedAt:t,...retain});
  b.set(doc(db,'partyEvents/partyCreated_room_1'),{schemaVersion:1,eventId:r.commandId,recruitmentId:'room',type:'partyCreated',commandId:r.commandId,
    commandRevision:1,audiencePlayerIds:['p0'],actorPlayerId:'p0',occurredAt:t});
  if(withClaim)b.set(doc(db,claimPath),claim(t));return b.commit();}
async function readyCertify(){await seed();await admin((b,db)=>{
  b.set(doc(db,'recruitments/room'),{...room(),currentCycleKey:'2026-09-30',cycleStartedAt:old,cycleEndsAt:now,continueTogether:true,currentCycleStatus:'completed'});
  b.set(doc(db,'recruitments/room/roster/current'),{schemaVersion:1,recruitmentId:'room',revision:1,entries:{p0:entry()}});
});}

  if(withClaim)b.set(doc(db,claimPath),claim(t));return b.commit();}
